"""Approved 2x2 baseline plus sampled HBM; no profiler or source changes."""
from pathlib import Path
import hashlib
import json
import os
import signal
import socket
import subprocess
import sys
import threading
import time
import urllib.error
import urllib.request

sys.path.insert(0, str(Path.cwd()))
from benchmarks.course_project.core import build_serve_command, load_model_config

root = Path("results/raw/homework1_final_baseline_hbm_20261015")
config_path = Path("configs/baseline/qwen35_0.8b.yaml")
config = load_model_config(config_path)
config = {**config, "prometheus_port": 39501, "tokenspeed_metrics_port": 39502}
config_path = Path(".tmp/homework1_final_baseline_config_20261015.json")
with config_path.open("x") as stream:
    json.dump(config, stream)
manifest_path = Path("results/raw/homework1_stage_b_20261015/source_manifest_r3.json")
full_source = json.loads(manifest_path.read_text())
def check_full_source():
    for name, expected in full_source.items():
        assert hashlib.sha256(Path(name).read_bytes()).hexdigest() == expected, name
check_full_source()
assert "TOKENSPEED_STAGE_B_ARM" not in os.environ
assert os.environ.get("ASCEND_RT_VISIBLE_DEVICES") == "2"
assert not any(key.startswith("TOKENSPEED_DIAG_") for key in os.environ)
for port in [18460, 18461, 39501, 39502]:
    with socket.socket() as sock:
        assert sock.connect_ex(("127.0.0.1", port)) != 0, f"port occupied: {port}"
device = subprocess.run(["npu-smi", "info"], capture_output=True, text=True, check=True)
mapping = subprocess.check_output(["npu-smi", "info", "-m"], text=True)
assert __import__("re").search(r"^\s*1\s+0\s+2\s+2\s+Ascend910\s*$", mapping, __import__("re").M)
idle = subprocess.check_output(["npu-smi", "info", "-t", "proc-mem", "-i", "1"], text=True)
assert idle.count("No process in device.") == 2
root.mkdir()
(root / "before_device.txt").write_text(device.stdout)
(root / "mapping.txt").write_text(mapping)
(root / "processes_before.txt").write_text(idle)
(root / "provenance").mkdir()
source_paths = [config_path, Path(__file__), Path("scripts/run_benchmark.py"),
                Path("benchmarks/course_project/core.py"), Path("benchmarks/course_project/runner.py"),
                Path("benchmarks/course_project/telemetry.py"),
                Path("python/tokenspeed/runtime/models/qwen3_5.py"),
                Path("python/tokenspeed/runtime/layers/layernorm.py"),
                Path("tokenspeed-kernel-npu/python/tokenspeed_kernel_npu/ops/layernorm.py")]
source_hashes = {}
for index, path in enumerate(source_paths):
    raw = path.read_bytes()
    with (root / "provenance" / f"{index}-{path.name}").open("xb") as stream:
        stream.write(raw)
    source_hashes[str(path)] = hashlib.sha256(raw).hexdigest()
with (root / "protocol.json").open("x") as stream:
    json.dump({"config": config, "source_sha256": source_hashes,
               "input_targets": [128, 1024], "output_targets": [32, 128],
               "concurrency": 1, "runs": 3, "requests_per_run": 8,
               "warmup_per_condition": 3, "logical_device": 2, "physical_npu": 1, "chip": 0, "source_manifest_sha256": hashlib.sha256(manifest_path.read_bytes()).hexdigest(),
               "hbm_sampling": "npu-smi full info; 1s delay after each query; sampled not instantaneous peak",
               "scope": "final-source HW1 baseline; NPU0 round snapshots from legacy helper are not target-resource evidence; HBM uses full device sampler physical1/chip0"}, stream, indent=2)

stop_sampling = threading.Event()
phase = "startup"


def sample_hbm() -> None:
    with (root / "hbm_samples.jsonl").open("x") as stream:
        while not stop_sampling.is_set():
            start = time.time_ns()
            result = subprocess.run(["npu-smi", "info"], capture_output=True, text=True, timeout=10)
            stream.write(json.dumps({"started_ns": start, "ended_ns": time.time_ns(),
                                     "phase": phase, "returncode": result.returncode,
                                     "stdout": result.stdout, "stderr": result.stderr}) + "\n")
            stream.flush()
            stop_sampling.wait(1)


sampler = threading.Thread(target=sample_hbm, daemon=True)
sampler.start()
service = None
try:
    with (root / "service.log").open("x") as log:
        service = subprocess.Popen(build_serve_command(config, host="127.0.0.1", port=18460),
                                   stdout=log, stderr=subprocess.STDOUT, start_new_session=True)
        with (root / "service_process.json").open("x") as stream:
            json.dump({"pid": service.pid, "pgid": os.getpgid(service.pid)}, stream)
        deadline = time.monotonic() + 180
        while True:
            assert service.poll() is None, "service exited before health"
            try:
                with urllib.request.urlopen("http://127.0.0.1:18460/health", timeout=2) as response:
                    if response.status == 200:
                        break
            except (urllib.error.URLError, TimeoutError):
                pass
            assert time.monotonic() < deadline, "health observation timeout; do not restart"
            time.sleep(1)
        (root / "loaded_device.txt").write_text(subprocess.check_output(["npu-smi", "info"], text=True))
        for output_tokens in [32, 128]:
            phase = f"measurement-out{output_tokens}"
            directory = root / f"out{output_tokens}"
            directory.mkdir()
            with (root / f"benchmark-out{output_tokens}.log").open("x") as bench_log:
                subprocess.run([sys.executable, "scripts/run_benchmark.py", str(config_path),
                                "--base-url", "http://127.0.0.1:18460/v1", "--concurrency", "1", "--input-tokens", "128", "1024",
                                "--output-tokens", str(output_tokens), "--num-requests", "8",
                                "--warmup-requests", "3", "--runs", "3", "--output-dir", str(directory)],
                               stdout=bench_log, stderr=subprocess.STDOUT, check=True)
        phase = "measurement-complete"
finally:
    phase = "teardown"
    if service is not None and service.poll() is None:
        assert os.getpgid(service.pid) == service.pid != os.getpgrp()
        os.killpg(service.pid, signal.SIGTERM)
        service.wait(timeout=60)
    stop_sampling.set()
    sampler.join(timeout=12)
    assert not sampler.is_alive(), "sampler did not stop"
    (root / "exit_device.txt").write_text(subprocess.check_output(["npu-smi", "info"], text=True))
(root / "processes_after.txt").write_text(subprocess.check_output(["npu-smi", "info", "-t", "proc-mem", "-i", "1"], text=True))
assert (root / "processes_after.txt").read_text().count("No process in device.") == 2
check_full_source()
print(json.dumps({"completed": True, "root": str(root), "service_exit": service.returncode}))
