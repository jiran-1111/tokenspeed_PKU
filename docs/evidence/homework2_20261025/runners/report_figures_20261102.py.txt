def export(fig, name, height):
    # Keep explicit panel and caption positions; pubfig's automatic tight_layout
    # overrides the reserved caption space in this report's fixed-size figures.
    fig.set_size_inches(180 / 25.4, height / 25.4, forward=True)

        path = OUTPUT / f"{name}.{suffix}"
        fig.savefig(path, format=suffix, dpi=240, facecolor="white")
        exported.append(path)
    plt.close(fig)

def relative_intervals(rank, kind):
    start = float(plot["native_window_us"][rank][0])
    items = plot["ranks"][rank]["intervals_us"][kind]
    return [(max(0, (a - start) / 1000), min(duration_ms, (b - start) / 1000)) for a, b in items]


def coverage(intervals):
    result = np.zeros(len(edges) - 1)
    for a, b in intervals:
        if b <= a:
            continue
        left = max(0, np.searchsorted(edges, a, side="right") - 1)
        right = min(len(result), np.searchsorted(edges, b, side="left") + 1)
        for i in range(left, right):
            result[i] += max(0, min(b, edges[i + 1]) - max(a, edges[i]))
    result /= np.diff(edges)
    assert np.all(result <= 1 + 1e-8)
    return np.clip(result, 0, 1)
