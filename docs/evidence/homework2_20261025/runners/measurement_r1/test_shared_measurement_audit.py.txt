def save(path: Path, value: object) -> str:
    path.write_text(json.dumps(value), encoding="utf-8")
    return hashlib.sha256(path.read_bytes()).hexdigest()

    def write(self) -> None:
        for name in ("config", "protocol", "inputs"):
            self.run[f"{name}_sha256"] = save(
                self.directory / f"{name}.json", self.run[name]
            )
        (self.directory / "requests.jsonl").write_text(
            "\n".join(map(json.dumps, self.requests)) + "\n", encoding="utf-8"
        )
        (self.directory / "hbm_samples.jsonl").write_text(
            "\n".join(map(json.dumps, self.run["hbm_samples"])) + "\n", encoding="utf-8"
        )
        save(self.path, self.run)

    def audit(self) -> dict:
        self.write()
        return audit_run(self.path, expected_tp=2, expected_protocol=self.protocol)

    def test_valid_shared_matrix_has_all_cells_and_recomputed_metrics(self) -> None:
        result = self.audit()
        self.assertEqual(result["execution_errors"], [])
        self.assertTrue(result["shared_performance_valid"])
        self.assertFalse(result["exclusive"])
        self.assertFalse(result["normal_performance_eligible"])
        self.assertTrue(result["hbm_coverage_complete"])
        self.assertEqual(len(result["cells"]), 4)
        cell = result["cells"][0]
        self.assertEqual(
            cell["throughput"], {"min": 400.0, "median": 400.0, "max": 400.0}
        )
        self.assertEqual(cell["ttft_ms"], {"mean": 1.0, "p50": 1.0, "p95": 1.0})
        self.assertEqual(cell["formal_request_count"], 4)
        self.assertEqual(result["profiler"]["status"], "not_requested")

    def test_old_functional_run_cannot_be_upgraded(self) -> None:
        self.run.pop("shared_performance_authorized")
        self.run.pop("measurement_scope")
        self.run["shared_functional_only"] = True
        self.run["guard"]["resource_mode"] = "shared_functional_only"
        result = self.audit()
        self.assertFalse(result["shared_performance_valid"])
        self.assertGreaterEqual(len(result["execution_errors"]), 3)

    def test_tp1_on_selected_chip_has_one_real_rank(self) -> None:
        self.run["config"]["tp_size"] = 1
        self.run["engine_cli_args"][
            self.run["engine_cli_args"].index("--world-size") + 1
        ] = "1"
        self.run["device"]["chip_logic_ids"].pop("1")
        self.run["guard"]["approved_background"].pop("1")
        for sample in self.run["guard"]["background_samples"]:
            sample["pids"].pop("1")
        for sample in self.run["process_samples"] + self.run["hbm_samples"]:
            sample["chips"].pop("1")
        release = json.loads(
            (self.directory / "shared_release.json").read_text(encoding="utf-8")
        )
        release["background"].pop("1")
        release["after_pids"].pop("1")
        save(self.directory / "shared_release.json", release)
        (self.directory / "rank_forward_events/rank1_pid102.forward.jsonl").unlink()
        self.write()
        result = audit_run(self.path, expected_tp=1, expected_protocol=self.protocol)
        self.assertTrue(result["shared_performance_valid"], result["execution_errors"])
        self.assertTrue(result["hbm_coverage_complete"], result["hbm_errors"])
        self.assertEqual(set(result["rank_forward_events"]), {"0"})

    def test_missing_cell_and_short_input_rejected_together(self) -> None:
        self.requests = [
            row
            for row in self.requests
            if (row["input_length"], row["concurrency"]) != (4, 2)
        ]
        self.requests[0]["input_ids"] = [7]
        result = self.audit()
        self.assertFalse(result["shared_performance_valid"])
        self.assertTrue(
            any("input IDs" in error for error in result["execution_errors"])
        )
        self.assertTrue(
            any("requests" in error for error in result["execution_errors"])
        )
        self.assertEqual(len(result["cells"]), 4)

    def test_resource_violation_and_cleanup_failures_are_retained(self) -> None:
        self.run["guard"]["violations"] = [
            {"type": "ValueError", "message": "unexpected process"}
        ]
        self.run["guard"]["own_children_released"] = False
        result = self.audit()
        self.assertFalse(result["execution_valid"])
        self.assertTrue(
            any("violations" in error for error in result["execution_errors"])
        )
        self.assertTrue(any("cleanup" in error for error in result["execution_errors"]))

    def test_missing_hbm_does_not_destroy_valid_execution_or_become_zero(self) -> None:
        self.run["hbm_samples"] = []
        self.run["sampler_errors"] = [{"error": "missing sample"}]
        result = self.audit()
        self.assertTrue(result["execution_valid"], result["execution_errors"])
        self.assertTrue(result["shared_performance_valid"])
        self.assertFalse(result["hbm_coverage_complete"])
        self.assertFalse(result["evidence_complete"])
        self.assertTrue(result["hbm_errors"])
        self.assertIsNone(result["batches"][0]["hbm_observed_max_mb_per_chip"])

    def test_failed_request_and_wrong_rank_source_rejected(self) -> None:
        self.requests[0]["status"] = "failed"
        self.requests[0]["error"] = "synthetic failure"
        (self.directory / "rank_forward_events/rank1_pid102.forward.jsonl").unlink()
        result = self.audit()
        self.assertFalse(result["execution_valid"])
        self.assertTrue(
            any("failed request" in error for error in result["execution_errors"])
        )
        self.assertTrue(any("rank" in error for error in result["execution_errors"]))

    def test_profiler_capture_is_separate_from_performance(self) -> None:
        self.run["profile_mode"] = "all-batches"
        self.run["profile"] = {
            "controls": [
                {"action": action, "acknowledged": True} for action in ("start", "stop")
            ],
            "traces": [],
        }
        profile = self.directory / "profile"
        profile.mkdir()
        for rank in (0, 1):
            trace = profile / f"hw2-TP{rank}.trace.json"
            sha = save(trace, {"traceEvents": []})
            self.run["profile"]["traces"].append(
                {
                    "path": trace.relative_to(self.directory).as_posix(),
                    "rank": rank,
                    "sha256": sha,
                }
            )
        result = self.audit()
        self.assertTrue(result["execution_valid"], result["execution_errors"])
        self.assertFalse(result["shared_performance_valid"])
        self.assertTrue(result["shared_profiler_valid"])
        self.assertEqual(result["profiler"]["status"], "valid_capture")

    def test_mapping_and_extra_round_log_are_verified(self) -> None:
        (self.directory / "before_mapping.txt").write_text(
            "1 0 4 4 Ascend910\n1 1 3 3 Ascend910\n", encoding="utf-8"
        )
        (self.directory / "rounds.jsonl").write_text("{}\n", encoding="utf-8")
        result = self.audit()
        self.assertFalse(result["execution_valid"])
        self.assertTrue(any("mapping" in error for error in result["execution_errors"]))
        self.assertTrue(any("rounds" in error for error in result["execution_errors"]))

    def test_missing_hbm_file_is_independent_of_execution(self) -> None:
        self.write()
        (self.directory / "hbm_samples.jsonl").unlink()
        result = audit_run(self.path, expected_tp=2, expected_protocol=self.protocol)
        self.assertTrue(result["execution_valid"], result["execution_errors"])
        self.assertFalse(result["hbm_coverage_complete"])
        self.assertTrue(result["hbm_errors"])
        self.assertTrue(
            all(
                batch["hbm_observed_max_mb_per_chip"] is None
                for batch in result["batches"]
            )
        )

    def test_declared_protocol_and_round_order_cannot_be_reduced(self) -> None:
        expected = dict(self.protocol, repetitions=3)
        self.run["batches"][0], self.run["batches"][1] = (
            self.run["batches"][1],
            self.run["batches"][0],
        )
        self.write()
        result = audit_run(self.path, expected_tp=2, expected_protocol=expected)
        self.assertFalse(result["execution_valid"])
        self.assertTrue(
            any("predeclared protocol" in error for error in result["execution_errors"])
        )
        self.assertTrue(
            any("batch order" in error for error in result["execution_errors"])
        )

    def test_bad_profiler_capture_does_not_gain_performance_status(self) -> None:
        self.run["profile_mode"] = "all-batches"
        result = self.audit()
        self.assertTrue(result["execution_valid"], result["execution_errors"])
        self.assertFalse(result["shared_performance_valid"])
        self.assertFalse(result["shared_profiler_valid"])
        self.assertEqual(result["profiler"]["status"], "invalid_capture")
        self.assertTrue(result["profiler"]["errors"])

    def test_malformed_run_fails_closed(self) -> None:
        self.run["guard"] = []
        self.write()
        result = audit_run(self.path, expected_tp=2)
        self.assertFalse(result["execution_valid"])
        self.assertTrue(result["execution_errors"])


if __name__ == "__main__":
    unittest.main()
