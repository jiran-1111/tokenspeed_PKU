"""Profiler-only setup: explicit MSTX and retained native input data."""


def configure_profiler(original):
    def create(backend, *, activities, with_stack, record_shapes):
        if not backend.is_npu:
            return original(
                backend,
                activities=activities,
                with_stack=with_stack,
                record_shapes=record_shapes,
            )
        experimental = backend.profiler._ExperimentalConfig(
            profiler_level=backend.profiler.ProfilerLevel.Level1,
            aic_metrics=backend.profiler.AiCMetrics.PipeUtilization,
            mstx=True,
            data_simplification=False,
        )
        return backend.profiler.profile(
            activities=[
                backend.activity_map[name]
                for name in activities
                if name in backend.activity_map
            ],
            with_stack=with_stack,
            record_shapes=record_shapes,
            profile_memory="MEM" in activities,
            experimental_config=experimental,
        )

    return create
