import importlib
import json
import os
import sys
import tempfile
import unittest
from contextlib import contextmanager
from decimal import Decimal
from pathlib import Path
from types import SimpleNamespace
from unittest.mock import patch

TASK = Path(__file__).resolve().parent
sys.path.insert(0, str(TASK))


def anchor(rank, seq, start, *, offset=1_000_000):
    name = f"HW2_SYNC|rank={rank}|seq={seq}|mono={start}|epoch={start + 2_000_000}"
    event = {
        "name": name,
        "rank": rank,
        "seq": seq,
        "m0": start,
        "m1": start + 10,
        "m2": start + 100,
        "m3": start + 110,
        "epoch0": start + 2_000_000,
        "epoch1": start + 2_000_010,
        "epoch2": start + 2_000_100,
        "epoch3": start + 2_000_110,
    }
    marker = {
        "name": name,
        "ph": "X",
        "pid": 10,
        "tid": 20,
        "ts": str(Decimal(start + 5 + offset) / 1000),
        "dur": "0.100",
    }
    return event, marker


class SharedProfileClockTests(unittest.TestCase):
    def setUp(self):
        self.assertTrue(
            (TASK / "shared_profile_clock.py").is_file(), "clock helper missing"
        )
        self.clock = importlib.import_module("shared_profile_clock")

    def evidence(self):
        pairs = [anchor(0, 1, 1000), anchor(0, 2, 2000)]
        return [event for event, _ in pairs], {
            "traceEvents": [marker for _, marker in pairs]
        }

    def test_forward_preserves_arguments_identity_and_records_every_call(self):
        calls, names, emitted = [], [], []

        @contextmanager
        def record_function(name):
            names.append(name)
            yield

        result = object()
        runner = SimpleNamespace(global_rank=1)

        def original(instance, *args, **kwargs):
            calls.append((instance, args, kwargs))
            return result

        wrapped = self.clock.observe_profiled_forward(original, emitted.append)
        fake_torch = SimpleNamespace(
            profiler=SimpleNamespace(record_function=record_function)
        )
        with patch.object(
            self.clock, "mstx_range", fake_torch.profiler.record_function
        ), patch.object(
            self.clock.time, "perf_counter_ns", side_effect=range(1200)
        ), patch.object(
            self.clock.time, "time_ns", side_effect=range(10000, 11200)
        ):
            for _ in range(300):
                self.assertIs(wrapped(runner, result, enabled=True), result)
        self.assertEqual(len(emitted), 300)
        self.assertEqual(len(names), 300)
        self.assertEqual(calls[-1], (runner, (result,), {"enabled": True}))
        self.assertEqual(emitted[-1]["seq"], 300)
        self.assertEqual([emitted[0][f"m{i}"] for i in range(4)], [0, 1, 2, 3])
        self.assertEqual(
            [emitted[0][f"epoch{i}"] for i in range(4)], [10000, 10001, 10002, 10003]
        )
        self.assertEqual(names[0], "HW2_SYNC|rank=1|seq=1|mono=0|epoch=10000")
        self.assertIsNone(emitted[0]["input_shape"])

    def test_forward_keeps_existing_audit_fields_and_only_reads_input_shape(self):
        emitted = []

        @contextmanager
        def record_function(name):
            yield

        inputs = SimpleNamespace(shape=(4, 2048))
        fake_torch = SimpleNamespace(
            profiler=SimpleNamespace(record_function=record_function)
        )
        with patch.object(
            self.clock, "mstx_range", fake_torch.profiler.record_function
        ):
            wrapped = self.clock.observe_profiled_forward(
                lambda *args, **kwargs: inputs, emitted.append
            )
            self.assertIs(
                wrapped(SimpleNamespace(global_rank=0), "context", inputs), inputs
            )
            self.assertIs(
                wrapped(SimpleNamespace(global_rank=0), input_ids=inputs), inputs
            )
        self.assertEqual(len(emitted), 2)
        for index, event in enumerate(emitted, 1):
            self.assertEqual(event["pid"], os.getpid())
            self.assertEqual(event["forward_call"], index)
            self.assertEqual(event["observed_ns"], event["m3"])
            self.assertEqual(event["input_shape"], [4, 2048])

    def test_forward_emits_final_bounds_and_preserves_original_exception(self):
        failure, emitted = RuntimeError("test failure"), []

        @contextmanager
        def record_function(name):
            yield

        def original(instance):
            raise failure

        fake_torch = SimpleNamespace(
            profiler=SimpleNamespace(record_function=record_function)
        )
        with patch.object(
            self.clock, "mstx_range", fake_torch.profiler.record_function
        ):
            with self.assertRaises(RuntimeError) as caught:
                self.clock.observe_profiled_forward(original, emitted.append)(
                    SimpleNamespace(global_rank=0)
                )
        self.assertIs(caught.exception, failure)
        self.assertEqual(len(emitted), 1)
        self.assertLessEqual(emitted[0]["m2"], emitted[0]["m3"])
        self.assertEqual(emitted[0]["status"], "failed")

    def test_emit_failure_does_not_replace_original_exception(self):
        failure = ValueError("original failure")

        @contextmanager
        def record_function(name):
            yield

        def original(instance):
            raise failure

        def emit(event):
            raise OSError("telemetry failed")

        fake_torch = SimpleNamespace(
            profiler=SimpleNamespace(record_function=record_function)
        )
        with patch.object(
            self.clock, "mstx_range", fake_torch.profiler.record_function
        ):
            with self.assertRaises(ValueError) as caught:
                self.clock.observe_profiled_forward(original, emit)(
                    SimpleNamespace(global_rank=0)
                )
        self.assertIs(caught.exception, failure)

    def test_alignment_uses_entry_and_exit_bounds_at_native_decimal_precision(self):
        events, trace = self.evidence()
        aligned = self.clock.align_trace(trace, events, rank=0, tolerance_ns=0)
        self.assertEqual(aligned["status"], "verified")
        self.assertEqual(aligned["offset_ns"], 1_000_000)
        self.assertEqual(aligned["offset_bounds_ns"], [999_995, 1_000_005])
        self.assertEqual(aligned["uncertainty_ns"], 5)
        self.assertEqual(aligned["anchor_count"], 2)
        self.assertEqual(aligned["perf_counter_span_ns"], [1000, 2110])
        self.assertEqual(aligned["sequence_span"], [1, 2])
        self.assertNotIn("formal_window", aligned)

    def test_alignment_accepts_path_without_float_rounding_epoch_timestamps(self):
        offset = 1_792_051_077_986_574_000
        pairs = [anchor(0, 1, 1_000, offset=offset), anchor(0, 2, 2_000, offset=offset)]
        with tempfile.TemporaryDirectory(dir=TASK) as directory:
            path = Path(directory) / "trace.json"
            content = json.dumps({"traceEvents": [marker for _, marker in pairs]})
            for _, marker in pairs:
                content = content.replace(json.dumps(marker["ts"]), marker["ts"])
            path.write_text(content)
            aligned = self.clock.align_trace(
                path, [event for event, _ in pairs], rank=0, tolerance_ns=0
            )
        self.assertEqual(aligned["offset_ns"], offset)
        self.assertEqual(aligned["uncertainty_ns"], 5)

    def test_alignment_rejects_missing_or_duplicate_marker(self):
        events, trace = self.evidence()
        for markers in (
            trace["traceEvents"][:1],
            trace["traceEvents"] + trace["traceEvents"][:1],
        ):
            with self.subTest(markers=len(markers)), self.assertRaisesRegex(
                ValueError, "marker"
            ):
                self.clock.align_trace(markers, events, rank=0, tolerance_ns=0)

    def test_alignment_rejects_wrong_rank_and_duplicate_log_evidence(self):
        events, trace = self.evidence()
        with self.assertRaisesRegex(ValueError, "rank"):
            self.clock.align_trace(trace, events, rank=1, tolerance_ns=0)
        with self.assertRaisesRegex(ValueError, "duplicate"):
            self.clock.align_trace(trace, events + events[:1], rank=0, tolerance_ns=0)

    def test_alignment_rejects_drift_beyond_explicit_tolerance(self):
        events, trace = self.evidence()
        trace["traceEvents"][1]["ts"] = str(Decimal(trace["traceEvents"][1]["ts"]) + 10)
        with self.assertRaisesRegex(ValueError, "offset|drift"):
            self.clock.align_trace(trace, events, rank=0, tolerance_ns=100)

    def test_alignment_rejects_clock_order_and_single_anchor(self):
        events, trace = self.evidence()
        events[0]["m2"] = events[0]["m0"] - 1
        with self.assertRaisesRegex(ValueError, "order"):
            self.clock.align_trace(trace, events, rank=0, tolerance_ns=0)
        with self.assertRaisesRegex(ValueError, "two|multiple"):
            self.clock.align_trace(
                trace["traceEvents"][1:], events[1:], rank=0, tolerance_ns=0
            )

    def test_alignment_rejects_mismatched_name_and_invalid_tolerance(self):
        events, trace = self.evidence()
        events[0]["m0"] += 1
        with self.assertRaisesRegex(ValueError, "name|marker"):
            self.clock.align_trace(trace, events, rank=0, tolerance_ns=0)
        for tolerance in (-1, True, 1.5):
            with self.subTest(tolerance=tolerance), self.assertRaisesRegex(
                ValueError, "tolerance"
            ):
                self.clock.align_trace(trace, [], rank=0, tolerance_ns=tolerance)


if __name__ == "__main__":
    unittest.main()
