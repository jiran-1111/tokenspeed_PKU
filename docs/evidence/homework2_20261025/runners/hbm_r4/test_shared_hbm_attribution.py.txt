import tempfile
import unittest
from copy import deepcopy
from pathlib import Path

# Reduced from the saved npu-smi 25.5.2 layout; names are never returned.
BORDER = "+===========================+===============+====================================================+"
HEADER = "| NPU     Chip              | Process id    | Process name             | Process memory(MB)      |"
ROWS = [
    "| 0       0                 | 20            | python                   | 112                     |",
    "| 1       0                 | 10            | python                   | 110                     |",
    "| 1       0                 | 30            | tokenspeedsch            | 2560                    |",
    "| 1       1                 | 10            | tokenspeedsch            | 2387                    |",
]
TABLE = "\n".join(["| npu-smi 25.5.2 |", BORDER, HEADER, BORDER, *ROWS, BORDER])


class ProcessMemoryTests(unittest.TestCase):
    def test_complete_table_returns_per_pid_memory_for_selected_card(self):
        self.assertTrue(
            Path(__file__).with_name("shared_hbm_attribution.py").is_file(),
            "process-memory parser is not implemented",
        )
        from shared_hbm_attribution import parse_process_memory

        self.assertEqual(
            parse_process_memory(TABLE, npu_id=1, chip_ids={0, 1}),
            {0: {10: 110, 30: 2560}, 1: {10: 2387}},
        )

    def test_target_without_rows_is_empty_only_with_a_complete_table(self):
        from shared_hbm_attribution import parse_process_memory

        self.assertEqual(parse_process_memory(TABLE, npu_id=3, chip_ids={0}), {0: {}})
        empty = "\n".join([HEADER, BORDER, "| No running processes found. |", BORDER])
        self.assertEqual(
            parse_process_memory(empty, npu_id=1, chip_ids={0, 1}), {0: {}, 1: {}}
        )

    def test_missing_truncated_malformed_and_duplicate_tables_are_rejected(self):
        from shared_hbm_attribution import parse_process_memory

        cases = [
            TABLE.replace(HEADER, "| unrecognized header |"),
            TABLE.rsplit("\n", 1)[0],
            TABLE + "\n" + HEADER + "\n" + BORDER,
            TABLE.replace("2387", "N/A"),
            TABLE.replace("2387", "-1"),
            TABLE.replace(ROWS[-1], ROWS[-1] + "\n" + ROWS[-1]),
            TABLE.replace(ROWS[-1], "| incomplete |"),
            TABLE.replace(ROWS[-1], "| 1 2 | 10 | python | 5 |"),
            "\n".join([HEADER, BORDER, BORDER]),
            "\n".join(
                [HEADER, BORDER, "| No running processes found |", ROWS[0], BORDER]
            ),
        ]
        for table in cases:
            with self.subTest(table=cases.index(table)), self.assertRaises(ValueError):
                parse_process_memory(table, npu_id=1, chip_ids={0, 1})

    def test_invalid_target_selection_is_rejected(self):
        from shared_hbm_attribution import parse_process_memory

        for card, chips in [(True, {0}), (-1, {0}), (1, set()), (1, {True}), (1, {2})]:
            with self.subTest(card=card, chips=chips), self.assertRaises(ValueError):
                parse_process_memory(TABLE, npu_id=card, chip_ids=chips)


def guard_sample(filename: str, owned: list[int]) -> dict:
    return {
        "label": "monitor",
        "observed_ns": 100,
        "raw_table": filename,
        "pids": {"0": [10, 30], "1": [10]},
        "identities": {"10": [7, "1000"], "30": [8, "2000"]},
        "exited_during_query": [],
        "owned_pids": owned,
        "memory": {
            "0": {"used_mb": 5900, "capacity_mb": 65536},
            "1": {"used_mb": 3000, "capacity_mb": 65536},
        },
    }


class GuardAttributionTests(unittest.TestCase):
    def setUp(self):
        self.directory = tempfile.TemporaryDirectory(dir=Path(__file__).parent)
        self.addCleanup(self.directory.cleanup)
        self.root = Path(self.directory.name)
        (self.root / "one.txt").write_text(TABLE, encoding="utf-8")
        self.run = {
            "device": {"npu_id": 1, "chip_logic_ids": {"0": 2, "1": 3}},
            "guard": {
                "approved_background": {"0": {"30": [8, "2000"]}, "1": {}},
                "background_samples": [guard_sample("one.txt", [10])],
            },
            # Deliberately contradictory; it is a separate observation.
            "process_samples": [{"owned_pids": [30]}],
            "engine_cli_args": ["DO_NOT_COPY_PROCESS_ARGUMENTS"],
            "env": {"secret": "DO_NOT_COPY_ENVIRONMENT"},
        }

    def analyze(self):
        import shared_hbm_attribution

        self.assertTrue(
            hasattr(shared_hbm_attribution, "analyze_guard_samples"),
            "guard-sample attribution is not implemented",
        )
        return shared_hbm_attribution.analyze_guard_samples(self.run, self.root)

    def test_own_background_and_chip_residual_are_separate_sampled_quantities(self):
        result = self.analyze()
        chip0 = result["samples"][0]["chips"]["0"]
        self.assertEqual(chip0["sampled_total_chip_used_mb"], 5900)
        self.assertEqual(chip0["sampled_own_process_mb"], 110)
        self.assertEqual(chip0["sampled_background_process_mb"], 2560)
        self.assertEqual(chip0["sampled_unknown_process_mb"], 0)
        self.assertEqual(chip0["sampled_unattributed_mb"], 3230)
        self.assertEqual(
            result["samples"][0]["chips"]["1"]["sampled_own_process_mb"], 2387
        )
        self.assertNotIn("DO_NOT_COPY", str(result))
        self.assertNotIn("tokenspeedsch", str(result))

    def test_pid_ownership_does_not_leak_between_guard_observations(self):
        sample = guard_sample("one.txt", [])
        sample["observed_ns"] = 200
        self.run["guard"]["background_samples"].append(sample)
        result = self.analyze()
        chip0 = result["samples"][1]["chips"]["0"]
        self.assertEqual(chip0["sampled_own_process_mb"], 0)
        self.assertEqual(chip0["sampled_unknown_process_mb"], 110)
        observed = result["sampled_ranges_mb"]["0"]["sampled_own_process_mb"]
        self.assertEqual(
            observed,
            {"sample_count": 2, "missing_sample_count": 0, "min": 0, "max": 110},
        )

    def test_missing_owned_list_remains_unknown_despite_other_samples(self):
        del self.run["guard"]["background_samples"][0]["owned_pids"]
        chip0 = self.analyze()["samples"][0]["chips"]["0"]
        self.assertIsNone(chip0["sampled_own_process_mb"])
        self.assertEqual(chip0["sampled_unknown_process_mb"], 110)

    def test_pid_reuse_and_exit_are_unknown_instead_of_approved_background(self):
        sample = self.run["guard"]["background_samples"][0]
        sample["identities"]["30"] = [8, "9999"]
        sample["exited_during_query"] = [10]
        chip0 = self.analyze()["samples"][0]["chips"]["0"]
        self.assertEqual(chip0["sampled_own_process_mb"], 0)
        self.assertEqual(chip0["sampled_background_process_mb"], 0)
        self.assertEqual(chip0["sampled_unknown_process_mb"], 2670)

    def test_missing_raw_file_does_not_create_zero_process_memory(self):
        self.run["guard"]["background_samples"].append(
            guard_sample("missing.txt", [10])
        )
        result = self.analyze()
        chip0 = result["samples"][1]["chips"]["0"]
        self.assertEqual(chip0["sampled_total_chip_used_mb"], 5900)
        for metric in [
            "sampled_own_process_mb",
            "sampled_background_process_mb",
            "sampled_unknown_process_mb",
            "sampled_unattributed_mb",
        ]:
            self.assertIsNone(chip0[metric])
        self.assertEqual(
            result["sampled_ranges_mb"]["0"]["sampled_own_process_mb"][
                "missing_sample_count"
            ],
            1,
        )

    def test_missing_chip_total_retains_process_samples_without_a_residual(self):
        sample = self.run["guard"]["background_samples"][0]
        del sample["memory"]["0"]
        chip0 = self.analyze()["samples"][0]["chips"]["0"]
        self.assertIsNone(chip0["sampled_total_chip_used_mb"])
        self.assertIsNone(chip0["sampled_unattributed_mb"])
        self.assertEqual(chip0["sampled_own_process_mb"], 110)

    def test_invalid_process_table_keeps_total_but_marks_attribution_missing(self):
        (self.root / "one.txt").write_text(
            TABLE.replace("2387", "N/A"), encoding="utf-8"
        )
        result = self.analyze()
        self.assertIn("invalid_process_table", result["samples"][0]["issues"])
        chip0 = result["samples"][0]["chips"]["0"]
        self.assertEqual(chip0["sampled_total_chip_used_mb"], 5900)
        self.assertIsNone(chip0["sampled_own_process_mb"])

    def test_recorded_pid_mismatch_disables_ownership_for_that_chip(self):
        self.run["guard"]["background_samples"][0]["pids"]["0"] = [10]
        chip0 = self.analyze()["samples"][0]["chips"]["0"]
        self.assertIsNone(chip0["sampled_own_process_mb"])
        self.assertIsNone(chip0["sampled_background_process_mb"])
        self.assertEqual(chip0["sampled_unknown_process_mb"], 2670)

    def test_missing_background_metadata_is_not_reported_as_zero(self):
        del self.run["guard"]["approved_background"]
        chip0 = self.analyze()["samples"][0]["chips"]["0"]
        self.assertIsNone(chip0["sampled_background_process_mb"])
        self.assertEqual(chip0["sampled_unknown_process_mb"], 2560)

    def test_inconsistent_negative_chip_residual_is_preserved_and_flagged(self):
        self.run["guard"]["background_samples"][0]["memory"]["0"]["used_mb"] = 100
        result = self.analyze()
        self.assertEqual(
            result["samples"][0]["chips"]["0"]["sampled_unattributed_mb"], -2570
        )
        self.assertIn(
            "chip_0_process_sum_exceeds_chip_total", result["samples"][0]["issues"]
        )

    def test_raw_path_must_remain_in_run_directory(self):
        self.run["guard"]["background_samples"][0]["raw_table"] = "../one.txt"
        result = self.analyze()
        self.assertIn("invalid_raw_table_path", result["samples"][0]["issues"])
        self.assertIsNone(result["samples"][0]["chips"]["0"]["sampled_own_process_mb"])

    def test_no_guard_samples_have_empty_ranges_not_zero_memory(self):
        self.run["guard"]["background_samples"] = []
        result = self.analyze()
        self.assertEqual(result["samples"], [])
        self.assertEqual(
            result["sampled_ranges_mb"]["0"]["sampled_own_process_mb"],
            {"sample_count": 0, "missing_sample_count": 0, "min": None, "max": None},
        )

    def test_analysis_does_not_modify_input(self):
        before = deepcopy(self.run)
        self.analyze()
        self.assertEqual(self.run, before)


if __name__ == "__main__":
    unittest.main()
