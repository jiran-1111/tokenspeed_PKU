OUT = Path(__file__).resolve().parent
WORKSPACE = OUT.parents[1]
EVIDENCE = OUT / "evidence"
RAW = WORKSPACE / "tokenspeed/results/raw/homework1_stage_b_20261015"


class FloatStorage:
    pass


class BFloat16Storage:
    pass


def rebuild(storage, offset, shape, stride, requires_grad, hooks, metadata=None):
    if requires_grad or len(shape) != len(stride) or any(x < 0 for x in shape + stride):
        raise ValueError("Unsupported tensor view")
    last = offset + sum((n - 1) * s for n, s in zip(shape, stride))
    if offset < 0 or (all(shape) and last >= storage.size):
        raise ValueError("Out-of-bounds tensor view")
    return np.ndarray(shape, dtype="<f4", buffer=storage, offset=offset * 4,
                      strides=tuple(s * 4 for s in stride))


def load_tensor_record(path):
    with zipfile.ZipFile(path) as archive:
        name = next(n for n in archive.namelist() if n.endswith("/data.pkl"))
        prefix = name[:-len("data.pkl")]
        cache = {}

        class Reader(pickle.Unpickler):
            def find_class(self, module, name):
                allowed = {("torch", "FloatStorage"): FloatStorage,
                           ("torch", "BFloat16Storage"): BFloat16Storage,
                           ("torch._utils", "_rebuild_tensor_v2"): rebuild,
                           ("collections", "OrderedDict"): OrderedDict}
                if (module, name) not in allowed:
                    raise ValueError(f"Unsupported global: {module}.{name}")
                return allowed[module, name]

            def persistent_load(self, pid):
                kind, dtype, key, location, count = pid
                if kind != "storage" or dtype not in (FloatStorage, BFloat16Storage) or location != "cpu":
                    raise ValueError("Only CPU float/BF16 storage is allowed")
                if key not in cache:
                    raw = archive.read(prefix + "data/" + key)
                    width = 4 if dtype is FloatStorage else 2
                    if len(raw) != count * width:
                        raise ValueError("Storage size mismatch")
                    cache[key] = (np.frombuffer(raw, dtype="<f4") if width == 4 else
                                  (np.frombuffer(raw, dtype="<u2").astype("<u4") << 16).view("<f4"))
                return cache[key]

        return Reader(io.BytesIO(archive.read(name))).load()


def bf16_round(values):
    bits = np.asarray(values, dtype="<f4").view("<u4")
    return ((bits + 0x7FFF + ((bits >> 16) & 1)) & 0xFFFF0000).view("<f4")


def metrics(actual, expected, *, state):
    difference = actual.astype(np.float64) - expected.astype(np.float64)
    atol, rtol = (0.0003, 0.01) if state else (0.002, 0.02)
    return dict(max_abs=float(np.max(np.abs(difference))),
                relative_l2=float(np.linalg.norm(difference) / np.linalg.norm(expected.astype(np.float64))),
                passed=bool(np.isfinite(actual).all() and np.isfinite(expected).all()
                            and np.all(np.abs(difference) <= atol + rtol * np.abs(expected.astype(np.float64)))))


started = time.perf_counter()
rows = list(csv.DictReader((EVIDENCE / "head_replay_r1/vectors.csv").open()))
replay = load_tensor_record(EVIDENCE / "head_replay_r1/replay_logits.pt")["replay_logits"]
assert replay.shape == (272, 248320) and len(rows) == 272
max_delta = 0.0
raw_hashes = json.loads((EVIDENCE / "captured_raw_sha256.json").read_text())
serialized_matches = 0
replay_bad, old_bad, resolved, introduced = [], [], [], []
for case in range(16):
    reference = load_tensor_record(RAW / f"cpu-fp32-eager-cached-threads1/case_{case:02d}.pt")
    for step in range(17):
        row = rows[case * 17 + step]
        expected = reference["cached_logits"][step]
        actual = replay[case * 17 + step]
        error = metrics(actual, expected, state=False)
        assert row["case_id"] == reference["case_id"] and int(row["step"]) == step
        assert int(actual.argmax()) == int(row["replay_argmax"])
        assert int(expected.argmax()) == int(row["reference_argmax"])
        delta = abs(error["relative_l2"] - float(row["replay_relative_l2"]))
        assert delta < 2e-13 and error["max_abs"] == float(row["replay_max_abs"])
        max_delta = max(max_delta, delta)
        before_wrong = row["original_argmax"] != row["reference_argmax"]
        after_wrong = row["replay_argmax"] != row["reference_argmax"]
        old_bad.append(before_wrong)
        replay_bad.append(after_wrong)
        resolved.append(before_wrong and not after_wrong)
        introduced.append(not before_wrong and after_wrong)
        old_path = RAW / f"engine_r3/case_{case:02d}/step_{step:02d}.pt"
        key = f"source/experiment_results/head_capture_r2/case_{case:02d}/step_{step:02d}.pt"
        serialized_matches += hashlib.sha256(old_path.read_bytes()).hexdigest() == raw_hashes[key]
assert (sum(old_bad), sum(replay_bad), sum(resolved), sum(introduced)) == (13, 9, 8, 4)

gdn_folder = EVIDENCE / "source/experiment_results/gdn_r1"
inputs = load_tensor_record(gdn_folder / "inputs.pt")
gdn_rows = []
for length in (64, 65, 127, 128, 129):
    data = load_tensor_record(gdn_folder / f"length_{length}.pt")
    published = json.loads((gdn_folder / f"length_{length}.json").read_text())
    state = inputs["initial"][0].swapaxes(-1, -2).copy()
    outputs = []
    for t in range(length):
        q, k, v = (inputs[name][0, t].astype(np.float32) for name in ("q", "k", "v"))
        q = q / np.sqrt(np.sum(q*q, axis=-1, keepdims=True, dtype=np.float32) + np.float32(1e-6))
        k = k / np.sqrt(np.sum(k*k, axis=-1, keepdims=True, dtype=np.float32) + np.float32(1e-6))
        state *= np.exp(inputs["g"][0, t])[:, None, None]
        residual = (v - np.einsum("hk,hkv->hv", k, state)) * inputs["beta"][0, t, :, None]
        state += k[:, :, None] * residual[:, None, :]
        outputs.append(np.einsum("hk,hkv->hv", q, state) * np.float32(128**-0.5))
    independent_output = np.stack(outputs)[None]
    independent_state = state.swapaxes(-1, -2)[None]
    ref_output_max = float(np.max(np.abs(independent_output - data["reference_output"])))
    ref_state_max = float(np.max(np.abs(independent_state - data["reference_state"])))
    assert ref_output_max < 2e-6 and ref_state_max < 2e-5
    checks = {}
    for kind in ("full", "segmented", "decode", "hybrid"):
        checks[f"{kind}_output"] = metrics(data[f"{kind}_output"], bf16_round(independent_output), state=False)
        checks[f"{kind}_state"] = metrics(data[f"{kind}_state"], independent_state, state=True)
    assert all(c["passed"] for c in checks.values())
    assert published["passed"] and published["untouched_slots_exact"]
    gdn_rows.append(dict(length=length, passed=True, independent_checks=checks,
                         numpy_vs_torch_reference_output_max_abs=ref_output_max,
                         numpy_vs_torch_reference_state_max_abs=ref_state_max))

summary = dict(replay_vectors=272, original_argmax_mismatches=sum(old_bad), replay_argmax_mismatches=sum(replay_bad),
               original_mismatches_resolved=sum(resolved), newly_introduced_mismatches=sum(introduced),
               replay_relative_l2_recheck_max_delta=max_delta, captured_serialized_sha_matches_original=serialized_matches,
               gdn_lengths=gdn_rows, elapsed_seconds=time.perf_counter()-started,
               torch_imported=False, new_forward_or_npu=False)
(OUT / "independent_numpy_audit.json").write_text(json.dumps(summary, indent=2)+"\n", encoding="utf-8")
print(json.dumps({k:v for k,v in summary.items() if k != "gdn_lengths"}))
print("Independent NumPy GDN recurrence: all 5 lengths / 40 output-state comparisons passed")
